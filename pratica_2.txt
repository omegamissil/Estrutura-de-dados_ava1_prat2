#include <stdio.h>
#include <stdlib.h>

#define N 6

struct fila 
{
    int *valor;
    int primeiro;
    int ultimo;
    int nItens;
};

void criarfila(struct fila *f) 
{
    f->valor = (int*) malloc(N * sizeof(int));
    f->primeiro = 0;
    f->ultimo = -1;
    f->nItens = 0;
}

void reiniciarfila(struct fila *f) 
{
    f->primeiro = 0;
    f->ultimo = -1;
    f->nItens = 0;
    printf("\n A fila foi reiniciada. \n\n");
}

void inserirfila(struct fila *f, int v) 
{
    if(f->ultimo == N)
        f->ultimo = -1;
    f->ultimo++;
    f->valor[f->ultimo] = v;
    f->nItens++;
}

int removerfila(struct fila *f) 
{
    int temp = f->valor[f->primeiro++];
    if(f->primeiro == N)
        f->primeiro = 0;
    f->nItens--;
    return temp;
}

int filavazia(struct fila *f) 
{
    return (f->nItens == 0);
}

int filacheia(struct fila *f) 
{
    return (f->nItens == N);
}

void mostrarfila(struct fila *f) 
{
    int cont, i;
    for (cont = 0, i = f->primeiro; cont < f->nItens; cont++)
    {
        printf("%d\t", f->valor[i]);
        i++;
        if (i == N)
            i = 0;
    }
    printf("\n\n");
}

void informarUso(struct fila *f) 
{
    int usada = f->nItens;       
    int livre = N - usada;     
    
    printf("\n");
    printf("ESTATISTICAS DA FILA:      \n");
    printf("Capacidade total : %d\n", N);
    printf("Espacos usados   : %d\n", usada);
    printf("Espacos livres   : %d\n", livre);
    printf("\n");
}

void main() 
{
    int opcao, valor;
    struct fila mfila;

    criarfila(&mfila);

    do 
    {
        printf("MENU: \n");
        printf("1 - Inserir na fila\n");
        printf("2 - Remover da fila\n");
        printf("3 - Mostrar fila\n");
        printf("4 - Informar uso (Usada e livre)\n");
        printf("5 - Reiniciar fila\n");
        printf("0 - Sair\n\n");
        printf("Opcao escolhida: ");
        scanf("%d", &opcao);

        switch (opcao) 
        {
            case 0:
                exit(0);
            case 1:
                if (filacheia(&mfila)) 
                {
                    printf("\nFila cheia\n\n");
                } 
                else 
                {
                    printf("\nValor a ser inserido na fila? ");
                    scanf("%d", &valor);
                    inserirfila(&mfila, valor);
                }
                break;
            case 2:
                if (filavazia(&mfila)) 
                {
                    printf("\nFila vazia\n\n");
                } 
                else 
                {
                    valor = removerfila(&mfila);
                    printf("\n%d removido\n\n", valor);
                }
                break;
            case 3:
                if (filavazia(&mfila)) 
                {
                    printf("\nFila vazia\n\n");
                } 
                else 
                {
                    printf("\nConteudo da fila: ");
                    mostrarfila(&mfila);
                }
                break;
            case 4:
                informarUso(&mfila);
                break;
            case 5:
                reiniciarfila(&mfila);
                break;
            default:
                printf("\nOpcao Invalida\n\n");
        }
    } while(1);
}
